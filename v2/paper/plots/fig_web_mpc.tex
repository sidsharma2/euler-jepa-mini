% One bar per row of the table, coloured by that row's 'color' key.
\newcommand{\mpcbars}[1]{%
  \pgfplotstableread{#1}\tbl
  \pgfplotstablegetrowsof{\tbl}\pgfmathtruncatemacro{\last}{\pgfplotsretval-1}
  \foreach \k in {0,...,\last}{%
    \pgfplotstablegetelem{\k}{color}\of{\tbl}\edef\cc{c-\pgfplotsretval}%
    \edef\temp{\noexpand\addplot[xbar, bar width=7pt, fill=\cc, draw=none,
      restrict expr to domain={\noexpand\coordindex}{\k:\k},
      error bars/.cd, x dir=both, x explicit, error bar style={black, line width=0.5pt}]
      table[y=y, x=excess, x error plus=err_plus, x error minus=err_minus] {#1};}\temp}}
